cmake_minimum_required(VERSION 3.16)
project(rocket6dof CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Eigen3 REQUIRED)
# Epigraph: a C++ modelling layer for second-order cone programs, here with the ECOS solver.
# Use -DEPIGRAPH_DIR=/path/to/Epigraph for a local checkout, otherwise it is fetched.
# ECOS's branch-and-bound file calls memcpy without including string.h; GCC 14 rejects that
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -include string.h")
include(FetchContent)
set(ENABLE_ECOS TRUE CACHE BOOL "" FORCE)
set(ENABLE_OSQP FALSE CACHE BOOL "" FORCE)
if(EPIGRAPH_DIR)
  add_subdirectory(${EPIGRAPH_DIR} epigraph)
else()
  FetchContent_Declare(epigraph GIT_REPOSITORY https://github.com/EmbersArc/Epigraph.git GIT_TAG master GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
  FetchContent_MakeAvailable(epigraph)
endif()
add_executable(rocket6dof src/main.cpp src/scvx.cpp)
target_include_directories(rocket6dof PRIVATE src)
target_link_libraries(rocket6dof epigraph Eigen3::Eigen)
target_compile_options(rocket6dof PRIVATE -O3 -Wall)  # no -march=native: Epigraph is built without it and Eigen types must match
add_executable(rocket6dof_mc src/mpc.cpp src/scvx.cpp)
target_include_directories(rocket6dof_mc PRIVATE src)
target_link_libraries(rocket6dof_mc epigraph Eigen3::Eigen)
target_compile_options(rocket6dof_mc PRIVATE -O3 -Wall)
